  `price` DECIMAL NOT NULL,
  `product_categorits_id` INT NOT NULL,
  `articul` VARCHAR(20) NOT NULL,
  `quality` INT NOT NULL DEFAULT 0,
  `created_at` DATE NOT NULL,
  PRIMARY KEY (`id`),
  UNIQUE INDEX `name_UNIQUE` (`name` ASC) VISIBLE,
  INDEX `product_categories_id_idx` (`product_categorits_id` ASC) VISIBLE,
  UNIQUE INDEX `articul_UNIQUE` (`articul` ASC) VISIBLE,
  CONSTRAINT `product_categories_id`

    REFERENCES `shopdb`.`product_categories` (`id`)

USE academy;

CREATE TABLE classes(
	id INT PRIMARY KEY AUTO_INCREMENT,
    class_name VARCHAR(50) NOT NULL,
    places TINYINT NOT NULL
);

CREATE TABLE student_groups(
	groups_id INT PRIMARY KEY AUTO_INCREMENT,
    group_name VARCHAR(50) NOT NULL,
    stud_count TINYINT,
    speciality VARCHAR(50) NOT NULL
);

CREATE TABLE teachers(
	id INT PRIMARY KEY AUTO_INCREMENT,
    full_name VARCHAR(100) NOT NULL,
    email VARCHAR(100) UNIQUE,
    phone VARCHAR(20) UNIQUE,
    created_at DATE
);
    
CREATE TABLE students(
	id INT PRIMARY KEY AUTO_INCREMENT,
    full_name VARCHAR(100) NOT NULL,
    email VARCHAR(100) UNIQUE,
    phone VARCHAR(20) UNIQUE,
    student_groups_id INT NOT NULL,
    birthday_date DATE,
    created_at DATE,
    
    FOREIGN KEY (student_groups_id)
    REFERENCES student_groups(groups_id)
);

CREATE TABLE subjects(
	id INT PRIMARY KEY AUTO_INCREMENT,
    sub_name VARCHAR(50) NOT NULL,
    hours DECIMAL NOT NULL,
    teacher_id INT NOT NULL,
    
    FOREIGN KEY (teacher_id)
    REFERENCES teachers(id)
);

CREATE TABLE lessions(
	id INT PRIMARY KEY AUTO_INCREMENT,
    theme_name VARCHAR(50) NOT NULL,
    stud_count TINYINT,
    subject_id INT NOT NULL,
    student_groups_id INT NOT NULL,
    teacher_id INT NOT NULL,
    date_at DATETIME,
    
    FOREIGN KEY (subject_id)
    REFERENCES subjects(id),
    FOREIGN KEY (student_groups_id)
    REFERENCES student_groups(groups_id),
    FOREIGN KEY (teacher_id)
    REFERENCES teachers(id)
);

CREATE TABLE student_grades(
	id INT PRIMARY KEY AUTO_INCREMENT,
    value_grade TINYINT NOT NULL,
    teacher_id INT NOT NULL,
    student_id INT NOT NULL,
    subject_id INT NOT NULL,
    lession_id INT NOT NULL,
    created_at DATETIME,
    
    FOREIGN KEY (teacher_id)
    REFERENCES teachers(id),
    FOREIGN KEY (student_id)
    REFERENCES students(id),
    FOREIGN KEY (subject_id)
    REFERENCES subjects(id),
    FOREIGN KEY (lession_id)
    REFERENCES lessions(id)
);

CREATE TABLE shedules(

-------------------------- sp_teacher_insert5 --------------------------

CREATE DEFINER=`root`@`localhost` PROCEDURE `sp_teacher_insert5`()
BEGIN
	DECLARE i INT default 1;
    DECLARE count INT default 1;
    SELECT Count(id)
    INTO i
    FROM teachers;
    SET count = i + 100;
    
	WHILE i < count DO
		INSERT INTO teachers(full_name, email, phone, created_at)
		VALUES(
			Concat("Teacher", i),
			Concat("teach_", i, "@gmail.com"),
			Concat("+7978", LPAD(i,2,"0")),
			curdate()
		);
		SET i = i + 1;
	END WHILE;
END

-------------------------- sp_subject_insert (procedure) --------------------------

CREATE DEFINER=`root`@`localhost` PROCEDURE `sp_subject_insert`()
BEGIN
	DECLARE i INT default 1;
	DECLARE count INT default 1;
    DECLARE t_id INT default 1;
    SELECT Count(id) INTO i FROM subjects;
    SET count = i + 100;
    
    WHILE i < count DO
		SET t_id = SELECT random_id_range();
        INSERT INTO subjects(sub_name, hours, teacher_id)
        
        VALUES(
			add_name_subject(),
            FLOOR(RAND() * 100),
            t_id
		);
		SET i = i + 1;
	END WHILE;
END

-------------------------- add_name_subject (function) --------------------------

CREATE FUNCTION `add_name_subject`()
RETURNS VARCHAR(50)
BEGIN
RETURN ELT(FLOOR(RAND() * 5) + 1, 
'Информатика', 'Биология', 'Физика', 'Химия', 'Базы данных');
END

-------------------------- random_id_range (function) --------------------------

CREATE DEFINER=`root`@`localhost` FUNCTION `random_id_range`(table_name VARCHAR(50)) RETURNS INT(11)
BEGIN
DECLARE t_id INT DEFAULT 1;
ELT(FLOOR(RAND() * 5) + 1, "teachers")
SELECT id 
	INTO t_id FROM tables_name ORDER BY RAND() LIMIT 1;
RETURN t_id;
END
